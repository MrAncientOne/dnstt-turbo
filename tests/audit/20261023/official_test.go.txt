var testBase32 = base32.StdEncoding.WithPadding(base32.NoPadding)

// testDomain is the zone these tests serve.
func testDomain(t *testing.T) dns.Name {
	t.Helper()
	d, err := dns.ParseName("t.example.com")
	if err != nil {
		t.Fatal(err)
	}
	return d
}

// testQuery builds a wire-valid query carrying raw (already clientID-
// prefixed) payload bytes in the QNAME, advertising payloadSize.
func testQuery(t *testing.T, domain dns.Name, id uint16, raw []byte, payloadSize uint16) *dns.Message {
	t.Helper()
	enc := testBase32.EncodeToString(raw)
	var labels [][]byte
	for len(enc) > 0 {
		n := 63
		if len(enc) < n {
			n = len(enc)
		}
		labels = append(labels, []byte(enc[:n]))
		enc = enc[n:]
	}
	name, err := dns.NewName(append(labels, domain...))
	if err != nil {
		t.Fatal(err)
	}
	return &dns.Message{
		ID:    id,
		Flags: 0x0100,
		Question: []dns.Question{
			{Name: name, Type: dns.RRTypeTXT, Class: dns.ClassIN},
		},
		Additional: []dns.RR{
			{Name: dns.Name{}, Type: dns.RRTypeOPT, Class: payloadSize, Data: []byte{}},
		},
	}
}

// clientPayload wraps packets the way the client QNAME path does:
// clientID + zero padding + length-prefixed packets.
func clientPayload(id turbotunnel.ClientID, packets ...[]byte) []byte {
	var b bytes.Buffer
	b.Write(id[:])
	b.WriteByte(224) // zero padding bytes
	for _, p := range packets {
		b.WriteByte(byte(len(p)))
		b.Write(p)
	}
	return b.Bytes()
}

// fakePacketConn is a scripted net.PacketConn: canned reads, recorded writes.
type fakePacketConn struct {
	reads  [][]byte
	writes [][]byte
	raddr  net.Addr
	err    error // permanent read error once reads exhaust
}

func (f *fakePacketConn) ReadFrom(p []byte) (int, net.Addr, error) {
	if len(f.reads) == 0 {
		if f.err != nil {
			return 0, nil, f.err
		}
		return 0, nil, errFakeClosed
	}
	n := copy(p, f.reads[0])
	f.reads = f.reads[1:]
	return n, f.raddr, nil
}

func (f *fakePacketConn) WriteTo(p []byte, _ net.Addr) (int, error) {
	f.writes = append(f.writes, append([]byte(nil), p...))
	return len(p), nil
}

func (f *fakePacketConn) Close() error                     { return nil }
func (f *fakePacketConn) LocalAddr() net.Addr              { return f.raddr }
func (f *fakePacketConn) SetDeadline(time.Time) error      { return nil }
func (f *fakePacketConn) SetReadDeadline(time.Time) error  { return nil }
func (f *fakePacketConn) SetWriteDeadline(time.Time) error { return nil }

type fakeAddr2 struct{}

func (fakeAddr2) Network() string { return "fake" }
func (fakeAddr2) String() string  { return "fake-addr" }

var errFakeClosed = errFake("closed")

type errFake string

func (e errFake) Error() string   { return string(e) }
func (e errFake) Timeout() bool   { return false }
func (e errFake) Temporary() bool { return false }

// Drives wire decoding -> response admission -> real response scheduler.
// No KCP/Noise handshake or data packet is supplied.
func TestAuditUnauthenticatedClientIDAllocation(t *testing.T) {
 for _, qsize := range []int{128} {
  t.Run(fmt.Sprintf("queue_%d",qsize),func(t *testing.T){
   tt:=turbotunnel.NewQueuePacketConn(turbotunnel.DummyAddr{},0)
   defer tt.Close()
   const n=8
   fc:=&fakePacketConn{raddr:fakeAddr2{}}
   for i:=0;i<n;i++ {
    var cid turbotunnel.ClientID
    binary.LittleEndian.PutUint64(cid[:], uint64(i+1))
    q:=testQuery(t,testDomain(t),uint16(i),clientPayload(cid),4096)
    wire,err:=q.WireFormat();if err!=nil {t.Fatal(err)}
    fc.reads=append(fc.reads,wire)
   }
   ch:=make(chan *record,n)
   runtime.GC()
   var before,after runtime.MemStats
   runtime.ReadMemStats(&before)
   _=recvLoop(testDomain(t),fc,tt,ch)
   close(ch)
   if err:=sendLoop(fc,tt,ch,computeMaxEncodedPayload(16384));err!=nil {t.Fatal(err)}
   runtime.GC(); runtime.ReadMemStats(&after)
   runtime.KeepAlive(tt)
   if len(fc.writes)!=n {t.Fatalf("wrote %d responses, want %d",len(fc.writes),n)}
   retained:=int64(after.HeapAlloc)-int64(before.HeapAlloc)
   t.Logf("queue=%d eight unauthenticated poll IDs retained=%d bytes",qsize,retained)
   if retained>2*1024*1024 {t.Errorf("eight empty polls retained %d bytes (>2MiB), with queue=%d",retained,qsize)}
  })
 }
}

func TestAuditAcceptedSmallResponseCapCanCarryConfiguredSegment(t *testing.T) {



 const serverMax=1232
 mtu:=computeMaxEncodedPayload(serverMax)-2

 for _,adv:=range []int{1232} {
  t.Run(fmt.Sprintf("advertised_%d",adv),func(t *testing.T){
   tt:=turbotunnel.NewQueuePacketConn(turbotunnel.DummyAddr{},0);defer tt.Close()
   var cid turbotunnel.ClientID;cid[0]=1
   // This is the segment size run()/acceptSessions configures, not an
   // oversized packet bypassing its configuration.
   pkt:=make([]byte,mtu)
   binary.LittleEndian.PutUint32(pkt[:4],1);pkt[4]=81
   binary.LittleEndian.PutUint32(pkt[20:24],uint32(len(pkt)-24))
   _,err:=tt.WriteTo(pkt,cid);if err!=nil {t.Fatal(err)}
   q:=testQuery(t,testDomain(t),1,clientPayload(cid),uint16(adv))
   resp,_:=responseFor(q,testDomain(t))
   if resp.Rcode()!=0 {t.Fatal("query was not accepted")}
   ch:=make(chan *record,1);ch<-&record{Resp:resp,Addr:fakeAddr2{},ClientID:cid};close(ch)
   fc:=&fakePacketConn{raddr:fakeAddr2{}}
   if err:=sendLoop(fc,tt,ch,computeMaxEncodedPayload(serverMax));err!=nil {t.Fatal(err)}
   if len(fc.writes)!=1 {t.Fatalf("responses=%d",len(fc.writes))}
   t.Logf("configured KCP MTU=%d advertised=%d wire=%d TC=%t",mtu,adv,len(fc.writes[0]),fc.writes[0][2]&2!=0)
   got,err:=dns.MessageFromWireFormat(fc.writes[0]);if err!=nil {t.Fatalf("accepted query got truncated unparseable response: %v",err)}
   payload,err:=dns.DecodeRDataTXT(got.Answer[0].Data);if err!=nil {t.Fatal(err)}
   if len(payload)!=mtu+2 {t.Fatalf("segment payload=%d want=%d",len(payload),mtu+2)}
  })
 }
